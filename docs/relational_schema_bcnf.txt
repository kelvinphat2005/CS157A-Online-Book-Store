Relational Schema 

Authors(author_id, name)
PK: author_id

Books(book_id, title, publisher, price, summary, stock, is_active, author_id)
PK: book_id
FK: author_id

Members(member_id, name, email, role, password_hash)
PK: member_id

Orders(order_id, member_id, delivery_address, status, total_cost)
PK: order_id
FK: member_id

Reviews(review_id, book_id, member_id, rating, review_text)
PK: review_id
FK: book_id, member_id

Orderline(order_id, book_id, quantity, price)
PK: (order_id, book_id)
FK: order_id
FK: book_id
